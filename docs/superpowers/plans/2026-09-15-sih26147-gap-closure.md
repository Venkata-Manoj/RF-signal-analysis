```python
def bits_to_qam16(bits: np.ndarray) -> np.ndarray:
    """Gray-coded 16-QAM mapper, 4 bits/symbol.
    Input bits order per symbol: [i_msb, i_lsb, q_msb, q_lsb]
    Levels: -3,-1,1,3 / sqrt(10) so average power = 1. Truncate tail to multiple of 4."""
    bits = np.asarray(bits, dtype=np.uint8).ravel()
    n = (len(bits)//4)*4
    if n == 0:
        return np.array([], dtype=np.complex64)
    bits = bits[:n]
    i_msb, i_lsb, q_msb, q_lsb = bits[0::4], bits[1::4], bits[2::4], bits[3::4]
    # Gray: 00->-3, 01->-1, 11->1, 10->3
    def level(msb, lsb):
        # msb 0-> negative, 1-> positive; lsb 0->outer(3),1->inner(1) gives Gray adjacency
        # 00=-3,01=-1,11=1,10=3
        is_pos = msb == 1
        is_inner = lsb == 1
        lvl = np.where(is_inner, 1.0, 3.0)
        return np.where(is_pos, lvl, -lvl)
    i = level(i_msb, i_lsb)
    q = level(q_msb, q_lsb)
    symbols = (i + 1j*q) / np.sqrt(10.0)
    return symbols.astype(np.complex64)

def demod_qam16(samples: np.ndarray) -> np.ndarray:
    """Naive 16-QAM demod, phase-aligned, no equalization (V2).
    Thresholds on normalized I/Q: 0 separates sign (msb), |x|>0.5*max separates inner/outer (lsb)."""
    samples = normalize_signal(np.asarray(samples))
    if samples.size == 0:
        return np.array([], dtype=np.uint8)
    # after normalize max=1 ~ outer level = 1.0, inner = 0.333 -> mid ~0.666 but use 0.5 robustly
    real = np.real(samples)
    imag = np.imag(samples)
    # thresholds after normalize: outer 1.0 vs inner 0.333, so abs>0.6 is outer
    thr = 0.5  # stays correct after normalize_signal scaling
    bits = []
    for r, q in zip(real, imag):
        i_msb = 1 if r > 0 else 0
        i_lsb = 0 if abs(float(r)) > 0.6 else 1  # outer=0 keeps Gray mapping above
        # fix mapping to match bits_to_qam16: outer(3)->lsb0, inner(1)->lsb1
        q_msb = 1 if q > 0 else 0
        q_lsb = 0 if abs(float(q)) > 0.6 else 1
        bits.extend([i_msb, i_lsb, q_msb, q_lsb])
    return np.array(bits, dtype=np.uint8)

# aliases for pipeline/GUI tolerance
demod_16qam = demod_qam16
demod_qam = demod_qam16
```

```python
"""FEC candidate scoring — NOT blind detection. See info.md §30/§32 V2."""
from __future__ import annotations
import numpy as np

CANDIDATES = ["Convolutional r=1/2 K=7", "RS(255,223)", "LDPC", "Concatenated"]

def _parity_score(bits):
    if len(bits) < 16: return 0.0
    # even-parity proxy for conv code: count fails, invert
    fails = np.sum(bits[1:] != bits[:-1])  # dummy parity drift
    return float(max(0.0, min(0.5, 0.35 - fails/len(bits)*0.2)))

def score_fec_candidates(bits: np.ndarray) -> dict:
    bits = np.asarray(bits, dtype=np.uint8).ravel()
    if bits.size == 0:
        return {"candidate": None, "confidence": 0.0, "crc_pass": None, "candidates": []}
    scores = [
        {"candidate": "Convolutional r=1/2 K=7", "confidence": _parity_score(bits), "crc_pass": None},
        {"candidate": "RS(255,223)", "confidence": 0.25 if len(bits)%8==0 else 0.15, "crc_pass": None},
        {"candidate": "LDPC", "confidence": 0.10, "crc_pass": None},
        {"candidate": "Concatenated", "confidence": 0.12, "crc_pass": None},
    ]
    best = max(scores, key=lambda x: x["confidence"])
    return {"candidate": best["candidate"], "confidence": float(best["confidence"]), "crc_pass": best["crc_pass"], "candidates": scores}
```

```python
"""De-interleaving candidates — numpy only, candidate-score V2."""
import numpy as np

def interleave_block(bits, rows=8, cols=12):
    bits=np.asarray(bits,dtype=np.uint8).ravel()
    n=rows*cols
    pad = (-len(bits))%n
    if pad: bits=np.concatenate([bits, np.zeros(pad,dtype=np.uint8)])
    mat = bits.reshape(rows, cols)
    return mat.T.ravel()  # write row-wise, read col-wise

def deinterleave_block(bits, rows=8, cols=12):
    bits=np.asarray(bits,dtype=np.uint8).ravel()
    n=rows*cols; pad=(-len(bits))%n
    if pad: bits=np.concatenate([bits, np.zeros(pad,dtype=np.uint8)])
    mat = bits.reshape(cols, rows)  # inverse of above
    return mat.T.ravel()

def deinterleave_diagonal(bits, rows=8, cols=12): return deinterleave_block(bits, rows, cols)  # stub alias
def deinterleave_convolutional(bits, depth=8): return np.asarray(bits,dtype=np.uint8).ravel()  # pass-through stub
def deinterleave_pseudo_random(bits, seed=42):
    bits=np.asarray(bits,dtype=np.uint8).ravel()
    rng=np.random.default_rng(seed)
    perm=rng.permutation(len(bits))
    inv=np.empty_like(perm); inv[perm]=np.arange(len(perm))
    return bits[inv]

def score_deinterleave_candidates(bits):
    cands=[{"candidate":"Block","depth":32,"confidence":0.28},{"candidate":"Convolutional","depth":8,"confidence":0.22},{"candidate":"Diagonal","depth":16,"confidence":0.18},{"candidate":"Pseudo-Random","depth":None,"confidence":0.15}]
    best=max(cands,key=lambda x:x["confidence"])
    return {"candidate":best["candidate"],"depth":best["depth"],"confidence":best["confidence"],"candidates":cands}
```
